    68→/** Summary cadangan bila backend belum mengirim objek summary. */
    69→function computeSummary(nodes: GlobalNodeT[]): GlobalResp["summary"] {
    70→  const regionCounts = new Map<string, number>();
    71→  const countries = new Set<string>();
    72→  let offices = 0;
    73→  let partners = 0;
    74→  for (const n of nodes) {
    75→    countries.add(n.countryCode);
    76→    regionCounts.set(n.region, (regionCounts.get(n.region) ?? 0) + 1);
    77→    if (n.kind === "KANTOR_PERWAKILAN") offices += 1;
    78→    if (n.kind === "MITRA_INTERNASIONAL") partners += 1;
    79→  }
    80→  return {
    81→    totalCountries: countries.size,
    82→    totalNodes: nodes.length,
    83→    offices,
    84→    partners,
    85→    regionCounts: [...regionCounts.entries()].map(([region, count]) => ({ region, count })),
    86→  };
    87→}

   231→export function GlobalNetwork() {
   232→  const [resp, setResp] = useState<GlobalResp | null>(null);
   233→  const [loading, setLoading] = useState(true);
   234→  const [error, setError] = useState<string | null>(null);
   235→
   236→  const [query, setQuery] = useState("");
   237→  const [kind, setKind] = useState("SEMUA");
   238→  const [region, setRegion] = useState("SEMUA");
   239→  const [selected, setSelected] = useState<GlobalNodeT | null>(null);
   240→  const [reloadKey, setReloadKey] = useState(0);
   241→
   242→  useEffect(() => {
   243→    let alive = true;
   244→    getGlobal()
   245→      .then((data) => {
   246→        if (alive) setResp(data);
   247→      })
   248→      .catch((e) => {
   249→        if (alive) setError(e instanceof Error ? e.message : "Gagal memuat jaringan global.");
   250→      })
   251→      .finally(() => {
   252→        if (alive) setLoading(false);
   253→      });
   254→    return () => {
   255→      alive = false;
   256→    };
   257→  }, [reloadKey]);
   258→
   259→  const retry = () => {
   260→    setLoading(true);
   261→    setError(null);
   262→    setReloadKey((k) => k + 1);
   263→  };
   264→
   265→  const nodes = resp?.nodes ?? [];
   266→  const summary = resp?.summary ?? computeSummary(nodes);
   267→
   268→  const regionCountMap = useMemo(() => {
   269→    const map: Record<string, number> = {};
   270→    for (const rc of summary.regionCounts) map[rc.region] = rc.count;
   271→    return map;
   272→  }, [summary]);
   273→
   274→  // daftar bendera unik (32 negara) untuk marquee
   275→  const flags = useMemo(() => {
   276→    const seen = new Set<string>();
   277→    const out: { code: string; flag: string; country: string }[] = [];
   278→    for (const n of nodes) {
   279→      if (seen.has(n.countryCode)) continue;
   280→      seen.add(n.countryCode);
   281→      out.push({ code: n.countryCode, flag: n.flag, country: n.country });
   282→    }
   283→    return out;
   284→  }, [nodes]);
   285→
   286→  const filtered = useMemo(() => {
   287→    const q = query.trim().toLowerCase();
   288→    return nodes.filter((n) => {
   289→      if (kind !== "SEMUA" && n.kind !== kind) return false;
   290→      if (region !== "SEMUA" && n.region !== region) return false;
   291→      if (q) {
   292→        const hay = `${n.country} ${n.city} ${n.orgName} ${n.title}`.toLowerCase();
   293→        if (!hay.includes(q)) return false;
   294→      }
   295→      return true;
   296→    });
   297→  }, [nodes, kind, region, query]);
   298→
   299→  return (
   300→    <section
   301→      id="jaringan-global"
   302→      className="relative scroll-mt-20 overflow-hidden bg-emerald-950 py-16 sm:py-20"
   303→    >
   304→      {/* ornamen latar gelap — selaras hero */}

   307→        <div className="absolute -right-32 bottom-0 h-96 w-96 rounded-full bg-amber-500/10 blur-3xl" />
   308→        <div className="absolute -left-32 top-1/3 h-80 w-80 rounded-full bg-emerald-800/40 blur-3xl" />
   309→      </div>
   310→
   311→      <div className="relative mx-auto w-full max-w-7xl px-4 sm:px-6 lg:px-8">
   312→        {/* Heading (varian dark dari SectionHeading — warna disesuaikan bg emerald-950) */}
   313→        <FadeIn className="mx-auto max-w-3xl text-center">
   314→          <p className="mb-3 inline-flex items-center gap-2 rounded-full border border-amber-400/40 bg-amber-400/10 px-3 py-1 text-xs font-semibold uppercase tracking-widest text-amber-300">
   315→            <span className="h-1.5 w-1.5 rounded-full bg-amber-400" aria-hidden />
   316→            Jaringan Global
   317→          </p>
   318→          <h2 className="font-display text-3xl font-extrabold text-white sm:text-4xl">
   319→            Layanan Sertifikasi di{" "}
   320→            <span className="bg-gradient-to-r from-amber-300 to-amber-500 bg-clip-text text-transparent">
   321→              32 Negara
   322→            </span>
   323→          </h2>
   324→          <p className="mt-3 text-base leading-relaxed text-emerald-100/80">
   325→            Kantor perwakilan dan mitra internasional PUSAKA tersebar lintas kawasan — temukan titik
   326→            layanan terdekat di negara Anda.
   327→          </p>
   328→        </FadeIn>
   329→
   330→        {/* 4 stat pill */}
   331→        <div className="mt-10 grid grid-cols-2 gap-4 lg:grid-cols-4">
   332→          {STAT_META.map((s, idx) => {

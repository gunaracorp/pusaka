   172→// Komponen utama — render <MemberPortal /> sekali di page.tsx
   173→// ============================================================
   174→export function MemberPortal() {
   175→  const { t, formatCurrency, formatDate } = useI18n();
   176→  const account = usePortalStore((s) => s.account);
   177→  const institution = usePortalStore((s) => s.institution);
   178→  const status = usePortalStore((s) => s.status);
   179→  const init = usePortalStore((s) => s.init);
   180→  const login = usePortalStore((s) => s.login);
   181→  const logout = usePortalStore((s) => s.logout);
   182→
   183→  const [open, setOpen] = useState(false);
   184→  const [applications, setApplications] = useState<PortalApplicationT[]>([]);
   185→  const [payments, setPayments] = useState<PaymentT[]>([]);
   186→  const [dataLoaded, setDataLoaded] = useState(false);
   187→
   188→  // Form login
   189→  const [email, setEmail] = useState("");
   190→  const [password, setPassword] = useState("");
   191→  const [showPassword, setShowPassword] = useState(false);
   192→  const [submitting, setSubmitting] = useState(false);
   193→
   194→  // Form ganti password
   195→  const [currentPassword, setCurrentPassword] = useState("");
   196→  const [newPassword, setNewPassword] = useState("");
   197→  const [confirmPassword, setConfirmPassword] = useState("");
   198→  const [pwError, setPwError] = useState<string | null>(null);
   199→  const [savingPassword, setSavingPassword] = useState(false);
   200→
   201→  // Guard mount (pola useSyncExternalStore — bebas set-state-in-effect)
   202→  const mounted = useSyncExternalStore(emptySubscribe, getClientSnapshot, getServerSnapshot);
   203→
   204→  // Reset error form ganti password saat dialog dibuka ulang
   205→  // (pola "adjust state during render" dari admin-login-dialog)
   206→  const [lastOpen, setLastOpen] = useState(open);
   207→  if (open !== lastOpen) {
   208→    setLastOpen(open);
   209→    if (open) setPwError(null);
   210→  }
   211→
   212→  const authed = mounted && status === "authed" && account !== null && institution !== null;
   213→
   214→  // MOUNT: init sesi + dengarkan event "pusaka:open-portal" dari navbar
   215→  useEffect(() => {
   216→    void init();
   217→    const onOpenPortal = () => setOpen(true);
   218→    window.addEventListener("pusaka:open-portal", onOpenPortal);
   219→    return () => window.removeEventListener("pusaka:open-portal", onOpenPortal);
   220→  }, [init]);
   221→
   222→  // DATA: refetch portalMe setiap dialog terbuka / status berubah authed
   223→  // (setState hanya di dalam .then/.finally — pola sama dengan news.tsx)
   224→  useEffect(() => {
   225→    if (!open || status !== "authed") return;
   226→    let alive = true;
   227→    portalMe()
   228→      .then((data) => {
   229→        if (!alive) return;
   230→        if (data) {
   231→          setApplications(data.applications);
   232→          setPayments(data.payments);
   233→        }
   234→      })
   235→      .catch(() => {
   236→        // diam — data lama / empty state tetap tampil
   237→      })
   238→      .finally(() => {
   239→        if (alive) setDataLoaded(true);
   240→      });
   241→    return () => {
   242→      alive = false;
   243→    };
   244→  }, [open, status]);
   245→
   246→  const handleLogin = useCallback(
   247→    async (e: FormEvent<HTMLFormElement>) => {
   248→      e.preventDefault();
   249→      if (submitting) return;
   250→      setSubmitting(true);
   251→      const ok = await login(email, password);
   252→      if (ok) {
   253→        setPassword("");
   254→        setShowPassword(false);
   255→        toast.success(t("portal.login.success"));
   256→      } else {
   257→        toast.error(t("portal.login.error"));
   258→      }
   259→      setSubmitting(false);
   260→    },
   261→    [email, password, login, submitting, t]
   262→  );
   263→
   264→  const fillDemo = useCallback(() => {
   265→    setEmail(DEMO_EMAIL);
   266→    setPassword(DEMO_PASSWORD);
   267→  }, []);
   268→
   269→  const handleCopy = useCallback(
   270→    (value: string) => {
   271→      const fail = () => toast.error(t("portal.misc.copyError"));
   272→      if (typeof navigator !== "undefined" && navigator.clipboard) {
   273→        navigator.clipboard
   274→          .writeText(value)
   275→          .then(() => toast.success(t("common.actions.copied")))
   276→          .catch(fail);
   277→      } else {
   278→        fail();
   279→      }
   280→    },
   281→    [t]
   282→  );
   283→
   284→  // "Verifikasi & Unduh": tutup dialog + arahkan ke anchor #verifikasi
   285→  const handleVerifyClick = useCallback(() => {
   286→    setOpen(false);
   287→    window.setTimeout(() => {
   288→      window.location.hash = "#verifikasi";
   289→    }, 250);
   290→  }, []);
   291→
   292→  // Invoice MENUNGGU → kirim kode ke halaman lacak (TrackSection autofill)
   293→  const handlePayNow = useCallback((trackingCode: string) => {
   294→    window.dispatchEvent(
   295→      new CustomEvent("pusaka:autofill-track", { detail: trackingCode })
   296→    );
   297→    setOpen(false);
   298→    toast.info(t("portal.pay.redirectTrack"));
   299→  }, [t]);
   300→
   301→  const handleLogout = useCallback(async () => {
   302→    await logout();
   303→    setApplications([]);
   304→    setPayments([]);
   305→    setDataLoaded(false);
   306→    setCurrentPassword("");
   307→    setNewPassword("");
   308→    setConfirmPassword("");
   309→    setPwError(null);
   310→    toast.success(t("portal.profile.logoutSuccess"));
   311→  }, [logout, t]);
   312→
   313→  const handleChangePassword = useCallback(
   314→    async (e: FormEvent<HTMLFormElement>) => {
   315→      e.preventDefault();
   316→      if (savingPassword) return;
   317→      if (!currentPassword) {
   318→        setPwError(t("portal.profile.emptyCurrent"));
   319→        return;
   320→      }
   321→      if (newPassword.length < 8) {
   322→        setPwError(t("portal.profile.minChars"));
   323→        return;
   324→      }
   325→      if (newPassword !== confirmPassword) {
   326→        setPwError(t("portal.profile.mismatch"));
   327→        return;
   328→      }
   329→      setPwError(null);
   330→      setSavingPassword(true);
   331→      try {
   332→        await portalChangePassword({ currentPassword, newPassword });
   333→        setCurrentPassword("");
   334→        setNewPassword("");
   335→        setConfirmPassword("");
   336→        toast.success(t("portal.profile.success"));
   337→      } catch {
   338→        toast.error(t("portal.profile.error"));
   339→      } finally {
   340→        setSavingPassword(false);
   341→      }
   342→    },
   343→    [confirmPassword, currentPassword, newPassword, savingPassword, t]
   344→  );
   345→
   346→  // ---------- Ringkasan pembayaran ----------
   347→  const totalPaid = payments
   348→    .filter((p) => p.status === "BERHASIL")
   349→    .reduce((sum, p) => sum + p.total, 0);
   350→  const activeDue = payments
   351→    .filter((p) => p.status === "MENUNGGU")
   352→    .reduce((sum, p) => sum + p.total, 0);
   353→
   354→  return (
   355→    <Dialog open={open} onOpenChange={setOpen}>
   356→      <DialogContent
   357→        className={`w-[min(94vw,760px)] max-h-[85vh] gap-5 overflow-y-auto ${SCROLLBAR_CLASS}`}
   358→      >
   359→        {!authed || !account || !institution ? (
   360→          // ================= VIEW GUEST / LOADING =================
   361→          <>
   362→            <DialogHeader className="items-center text-center">
   363→              <div
   364→                className="mx-auto mb-1 flex h-14 w-14 items-center justify-center rounded-full bg-gradient-to-br from-emerald-600 to-amber-500 text-white shadow-md shadow-emerald-900/20"
   365→                aria-hidden
   366→              >
   367→                <UserRound className="h-7 w-7" />
   368→              </div>
   369→              <DialogTitle className="text-center font-display text-xl">
   370→                {t("portal.login.title")}
   371→              </DialogTitle>
   372→              <DialogDescription className="text-center">
   373→                {t("portal.login.subtitle")}
   374→              </DialogDescription>
   375→            </DialogHeader>
   376→
   377→            {mounted && status === "loading" ? (
   378→              <div className="flex flex-col items-center gap-3 py-10" role="status" aria-live="polite">
   379→                <Loader2
   380→                  className="h-8 w-8 animate-spin text-emerald-600 dark:text-emerald-400"
   381→                  aria-hidden
   382→                />
   383→                <p className="text-sm text-muted-foreground">{t("common.actions.loading")}</p>
   384→              </div>
   385→            ) : (
   386→              <>
   387→                <form onSubmit={handleLogin} className="space-y-4" noValidate>
   388→                  <div className="space-y-2">
   389→                    <Label htmlFor="portal-email">{t("portal.login.emailLabel")}</Label>
   390→                    <Input
   391→                      id="portal-email"
   392→                      name="email"
   393→                      type="email"
   394→                      autoComplete="email"
   395→                      placeholder={t("portal.login.emailPlaceholder")}
   396→                      value={email}
   397→                      onChange={(e) => setEmail(e.target.value)}
   398→                      disabled={submitting}
   399→                      required
   400→                    />
   401→                  </div>
   402→
   403→                  <div className="space-y-2">
   404→                    <Label htmlFor="portal-password">{t("portal.login.passwordLabel")}</Label>
   405→                    <div className="relative">
   406→                      <Input
   407→                        id="portal-password"
   408→                        name="password"
   409→                        type={showPassword ? "text" : "password"}
   410→                        autoComplete="current-password"
   411→                        placeholder="••••••••"
   412→                        value={password}
   413→                        onChange={(e) => setPassword(e.target.value)}
   414→                        className="pr-10"
   415→                        disabled={submitting}
   416→                        required
   417→                      />
   418→                      <button
   419→                        type="button"
   420→                        onClick={() => setShowPassword((v) => !v)}
   421→                        className="absolute right-2 top-1/2 flex h-7 w-7 -translate-y-1/2 items-center justify-center rounded-md text-muted-foreground transition-colors hover:bg-emerald-500/10 hover:text-emerald-700 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-600 dark:hover:text-emerald-300"
   422→                        aria-label={
   423→                          showPassword
   424→                            ? t("portal.login.hidePassword")
   425→                            : t("portal.login.showPassword")
   426→                        }
   427→                        aria-pressed={showPassword}
   428→                      >
   429→                        {showPassword ? (
   430→                          <EyeOff className="h-4 w-4" aria-hidden />
   431→                        ) : (
   432→                          <Eye className="h-4 w-4" aria-hidden />
   433→                        )}
   434→                      </button>
   435→                    </div>
   436→                  </div>
   437→
   438→                  <Button
   439→                    type="submit"
   440→                    disabled={submitting}
   441→                    className="h-11 w-full gap-2 bg-emerald-700 text-white hover:bg-emerald-800"
   442→                  >
   443→                    {submitting ? (
   444→                      <>
   445→                        <Loader2 className="h-4 w-4 animate-spin" aria-hidden />
   446→                        {t("portal.login.submitting")}
   447→                      </>
   448→                    ) : (
   449→                      <>
   450→                        <LogIn className="h-4 w-4" aria-hidden />
   451→                        {t("portal.login.submit")}
   452→                      </>
   453→                    )}
   454→                  </Button>
   455→                </form>
   456→
   457→                {/* Hint akun demo */}
   458→                <div className="rounded-lg border border-dashed border-amber-500/40 bg-amber-500/10 p-3">
   459→                  <div className="flex items-start gap-2">
   460→                    <KeyRound
   461→                      className="mt-0.5 h-4 w-4 shrink-0 text-amber-600 dark:text-amber-400"
   462→                      aria-hidden
   463→                    />
   464→                    <div className="min-w-0 flex-1">
   465→                      <p className="text-xs font-medium text-amber-800 dark:text-amber-200">
   466→                        {t("portal.login.demoHint")}
   467→                      </p>
   468→                      <div className="mt-1.5 flex flex-wrap items-center justify-between gap-2">
   469→                        <code className="min-w-0 break-all font-mono text-[11px] text-amber-900 dark:text-amber-100">
   470→                          {DEMO_EMAIL} / {DEMO_PASSWORD}
   471→                        </code>
   472→                        <Button
   473→                          type="button"
   474→                          variant="ghost"
   475→                          size="sm"
   476→                          className="h-7 shrink-0 gap-1.5 px-2 text-xs text-amber-700 hover:bg-amber-500/20 hover:text-amber-800 dark:text-amber-300 dark:hover:text-amber-200"
   477→                          onClick={fillDemo}
   478→                          aria-label={`${t("portal.login.useAccount")} ${DEMO_EMAIL}`}
   479→                        >
   480→                          {t("portal.login.useAccount")}
   481→                        </Button>
   482→                      </div>
   483→                    </div>
   484→                  </div>
   485→                </div>
   486→              </>
   487→            )}
   488→          </>
   489→        ) : (
   490→          // ================= VIEW AUTHED =================
   491→          <>
   492→            {/* Header institusi */}
   493→            <div className="rounded-xl border border-emerald-600/20 bg-gradient-to-br from-emerald-500/10 via-transparent to-amber-500/10 p-4 dark:border-emerald-500/30">
   494→              <div className="flex flex-wrap items-start justify-between gap-3">
   495→                <div className="flex min-w-0 items-start gap-3">
   496→                  <span
   497→                    className="flex h-11 w-11 shrink-0 items-center justify-center rounded-full bg-gradient-to-br from-emerald-600 to-amber-500 font-display text-lg font-black text-white shadow-sm"
   498→                    aria-hidden
   499→                  >
   500→                    {institution.name.charAt(0)}
   501→                  </span>
   502→                  <div className="min-w-0">
   503→                    <DialogTitle className="font-display text-lg font-bold leading-tight text-foreground">
   504→                      {institution.name}
   505→                    </DialogTitle>
   506→                    <DialogDescription className="sr-only">
   507→                      {t("portal.login.title")}
   508→                    </DialogDescription>
   509→                    <p className="mt-0.5 font-mono text-xs text-muted-foreground">
   510→                      {institution.certNumber}
   511→                    </p>
   512→                  </div>
   513→                </div>
   514→                <div className="flex flex-wrap items-center gap-1.5">
   515→                  <Badge
   516→                    variant="outline"
   517→                    className={cn(
   518→                      TYPE_META[institution.type]?.badgeClass ??
   519→                        "border-border bg-muted text-foreground"
   520→                    )}
   521→                  >
   522→                    {institution.type}
   523→                  </Badge>
   524→                  <Badge variant="outline" className={TIER_META[institution.tier]?.badgeClass}>
   525→                    <span
   526→                      className={cn(
   527→                        "h-1.5 w-1.5 rounded-full",
   528→                        TIER_META[institution.tier]?.dotClass ?? "bg-muted-foreground"

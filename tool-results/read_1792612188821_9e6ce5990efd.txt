    67→/** Terjemahan kunci dinamis; label meta jadi fallback bila kunci kamus tidak ada. */
    68→function metaT(
    69→  t: (path: string, vars?: Record<string, string | number>) => string,
    70→  key: string,
    71→  fallback: string
    72→): string {
    73→  const value = t(key);
    74→  return value === key ? fallback : value;
    75→}

    77→/** Summary cadangan bila backend belum mengirim objek summary. */
    78→function computeSummary(nodes: GlobalNodeT[]): GlobalResp["summary"] {
    79→  const regionCounts = new Map<string, number>();
    80→  const countries = new Set<string>();
    81→  let offices = 0;
    82→  let partners = 0;
    83→  for (const n of nodes) {
    84→    countries.add(n.countryCode);
    85→    regionCounts.set(n.region, (regionCounts.get(n.region) ?? 0) + 1);
    86→    if (n.kind === "KANTOR_PERWAKILAN") offices += 1;
    87→    if (n.kind === "MITRA_INTERNASIONAL") partners += 1;
    88→  }
    89→  return {
    90→    totalCountries: countries.size,
    91→    totalNodes: nodes.length,
    92→    offices,
    93→    partners,
    94→    regionCounts: [...regionCounts.entries()].map(([region, count]) => ({ region, count })),
    95→  };
    96→}

   252→export function GlobalNetwork() {
   253→  const { t, formatNumber } = useI18n();
   254→  const [resp, setResp] = useState<GlobalResp | null>(null);
   255→  const [loading, setLoading] = useState(true);
   256→  const [error, setError] = useState<string | null>(null);
   257→
   258→  const [query, setQuery] = useState("");
   259→  const [kind, setKind] = useState("SEMUA");
   260→  const [region, setRegion] = useState("SEMUA");
   261→  const [selected, setSelected] = useState<GlobalNodeT | null>(null);
   262→  const [reloadKey, setReloadKey] = useState(0);
   263→
   264→  // Ref agar fetch awal tidak perlu t sebagai dependency (identitas t berubah saat ganti bahasa).
   265→  const tRef = useRef(t);
   266→  useEffect(() => {
   267→    tRef.current = t;
   268→  }, [t]);
   269→
   270→  useEffect(() => {
   271→    let alive = true;
   272→    getGlobal()
   273→      .then((data) => {
   274→        if (alive) setResp(data);
   275→      })
   276→      .catch((e) => {
   277→        if (alive) setError(e instanceof Error ? e.message : tRef.current("globalNetwork.loadError"));
   278→      })
   279→      .finally(() => {
   280→        if (alive) setLoading(false);
   281→      });
   282→    return () => {
   283→      alive = false;
   284→    };
   285→  }, [reloadKey]);
   286→
   287→  const retry = () => {
   288→    setLoading(true);
   289→    setError(null);
   290→    setReloadKey((k) => k + 1);
   291→  };
   292→
   293→  const nodes = resp?.nodes ?? [];
   294→  const summary = resp?.summary ?? computeSummary(nodes);
   295→
   296→  const regionCountMap = useMemo(() => {
   297→    const map: Record<string, number> = {};
   298→    for (const rc of summary.regionCounts) map[rc.region] = rc.count;
   299→    return map;
   300→  }, [summary]);
   301→
   302→  // daftar bendera unik (32 negara) untuk marquee
   303→  const flags = useMemo(() => {
   304→    const seen = new Set<string>();
   305→    const out: { code: string; flag: string; country: string }[] = [];
   306→    for (const n of nodes) {
   307→      if (seen.has(n.countryCode)) continue;
   308→      seen.add(n.countryCode);
   309→      out.push({ code: n.countryCode, flag: n.flag, country: n.country });
   310→    }
   311→    return out;
   312→  }, [nodes]);
   313→
   314→  const filtered = useMemo(() => {
   315→    const q = query.trim().toLowerCase();
   316→    return nodes.filter((n) => {
   317→      if (kind !== "SEMUA" && n.kind !== kind) return false;
   318→      if (region !== "SEMUA" && n.region !== region) return false;
   319→      if (q) {
   320→        const hay = `${n.country} ${n.city} ${n.orgName} ${n.title}`.toLowerCase();
   321→        if (!hay.includes(q)) return false;
   322→      }
   323→      return true;
   324→    });
   325→  }, [nodes, kind, region, query]);
   326→
   327→  return (
   328→    <section
   329→      id="jaringan-global"
   330→      className="relative scroll-mt-20 overflow-hidden bg-emerald-950 py-16 sm:py-20"
   331→    >
   332→      {/* ornamen latar gelap — selaras hero */}
   333→      <div className="pointer-events-none absolute inset-0" aria-hidden>
   334→        <div className="absolute -top-24 left-1/2 h-[420px] w-[860px] max-w-[140vw] -translate-x-1/2 rounded-full bg-emerald-900/50 blur-3xl" />
   335→        <div className="absolute -right-32 bottom-0 h-96 w-96 rounded-full bg-amber-500/10 blur-3xl" />
   336→        <div className="absolute -left-32 top-1/3 h-80 w-80 rounded-full bg-emerald-800/40 blur-3xl" />
   337→      </div>
   338→
   339→      <div className="relative mx-auto w-full max-w-7xl px-4 sm:px-6 lg:px-8">
   340→        {/* Heading (varian dark dari SectionHeading — warna disesuaikan bg emerald-950) */}
   341→        <FadeIn className="mx-auto max-w-3xl text-center">
   342→          <p className="mb-3 inline-flex items-center gap-2 rounded-full border border-amber-400/40 bg-amber-400/10 px-3 py-1 text-xs font-semibold uppercase tracking-widest text-amber-300">
   343→            <span className="h-1.5 w-1.5 rounded-full bg-amber-400" aria-hidden />
   344→            {t("globalNetwork.eyebrow")}
   345→          </p>
   346→          <h2 className="font-display text-3xl font-extrabold text-white sm:text-4xl">
   347→            {t("globalNetwork.titleBefore")}{" "}
   348→            <span className="bg-gradient-to-r from-amber-300 to-amber-500 bg-clip-text text-transparent">
   349→              {t("globalNetwork.titleHighlight")}
   350→            </span>
   351→          </h2>

   573→                        className="shrink-0 border-amber-400/60 bg-amber-400/10 text-[10px] font-bold uppercase tracking-wider text-amber-300"

   626→          </div>

   628→
   629→        {!loading && !error && filtered.length > 0 ? (

   631→            {t("globalNetwork.showing", {

   637→
   638→        {loading ? (

   640→            <Loader2 className="animate-spin" /> {t("globalNetwork.loadingAria")}
   641→          </span>
   642→        ) : null}
   643→      </div>
   644→
   645→      <NodeDetailDialog node={selected} onClose={() => setSelected(null)} />
   646→    </section>

    68→    68→function metaT(
    69→    69→  t: (path: string, vars?: Record<string, string | number>) => string,
    70→    70→  key: string,
    71→    71→  fallback: string
    72→    72→): string {

    74→    74→  return value === key ? fallback : value;

    77→    77→/** Summary cadangan bila backend belum mengirim objek summary. */
    78→    78→function computeSummary(nodes: GlobalNodeT[]): GlobalResp["summary"] {
    79→    79→  const regionCounts = new Map<string, number>();
    80→    80→  const countries = new Set<string>();
    81→    81→  let offices = 0;
    82→    82→  let partners = 0;
    83→    83→  for (const n of nodes) {
    84→    84→    countries.add(n.countryCode);
    85→    85→    regionCounts.set(n.region, (regionCounts.get(n.region) ?? 0) + 1);
    86→    86→    if (n.kind === "KANTOR_PERWAKILAN") offices += 1;
    87→    87→    if (n.kind === "MITRA_INTERNASIONAL") partners += 1;
    88→    88→  }
    89→    89→  return {
    90→    90→    totalCountries: countries.size,
    91→    91→    totalNodes: nodes.length,
    92→    92→    offices,
    93→    93→    partners,
    94→    94→    regionCounts: [...regionCounts.entries()].map(([region, count]) => ({ region, count })),
    95→    95→  };
    96→    96→}
